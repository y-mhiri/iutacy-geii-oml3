\rappelhead{Le symbole $\sum$}

Pour une suite de nombres $(u_k)$ et deux entiers $m\leq n$, on note
$$
    \sum_{k=m}^{n} u_k = u_m + u_{m+1} + \cdots + u_n,
$$
où $k$ est l'\highlight{indice de sommation} (une variable muette : elle n'apparaît pas dans le résultat) et $m,n$ les bornes. Par exemple, $\sum_{k=1}^{4} k^2 = 1^2+2^2+3^2+4^2 = 1+4+9+16 = 30$.

\rappelhead{Calcul intégral}

\begin{definition}
    Soit une fonction $f$ définie sur un intervalle réel $[a,b]$ et une primitive $F$ de $f$, c'est-à-dire telle que
    $F^\prime = f$. L'intégrale de $f$ entre $a$ et $b$ s'écrit alors
    $$
    \int_{a}^{b} f(t)\d t = \left[\ F(t)\ \right]_b^a = F(b) - F(a).
    $$
\end{definition}

\begin{proposition}[Relation de Chasles]
    Soit $f$ une fonction définie sur $\R$ et trois réels $a,b,c$, alors
    $$
    \int_{a}^{c} f(t)\d t = \int_{a}^{b} f(t)\d t + \int_{b}^{c} f(t)\d t
    $$
\end{proposition}

\begin{proposition}[Linéarité de l'intégrale]
    Soit $f$ et $g$ deux fonctions définies sur $\R$, $a,b, \lambda\in\R$, alors
    $$
    \begin{aligned}
        \int_a^b \left(f(t) + g(t)\right)\d t &= \int_a^b f(t) \d t + \int_{a}^{b} g(t) \d t \\
        \int_a^b \lambda f(t)\d t &= \lambda \int_a^b f(t) \d t.
    \end{aligned}
    $$
\end{proposition}

\begin{remark}
    \red{ATTENTION :} Dans le cas général le produit des intégrales n'est pas égale à l'intégral du produit,
    $$
        \int_a^b f(t)g(t) \d t \neq \left(\int_a^b f(t) \d t\right)\left(\int_a^b g(t)\d t\right).
    $$
\end{remark}

\begin{proposition}[Parité et intégrale]
    Soit $f$ une fonction définie sur $[-a,a]$ ($a>0$).
    \begin{itemize}
        \item Si $f$ est \highlight{paire}, c'est à dire $f(-t) = f(t)$, alors
        $$
            \int_{-a}^{a} f(t)\d t = 2\int_{0}^{a} f(t)\d t.
        $$
        \item Si $f$ est \highlight{impaire}, $f(-t) = -f(t)$, alors
        $$
            \int_{-a}^{a} f(t)\d t = 0.
        $$
    \end{itemize}
\end{proposition}

\begin{remark}
   La propriété précédente illustre le fait que pour une fonction impaire les aires algébriques de part et d'autre de $0$ s'annulent exactement. Pour une fonction paire, elles se dupliquent.
\end{remark}

\section{Fonctions périodiques}

\begin{definition}[Fonction périodique]
    Soit une fonction $f : \R \to \R$ (ou $\C$). On dit que $f$ est \highlight{$T$-périodique}, ou périodique de période $T$, s'il existe un réel $T > 0$ tel que
    $$
        \forall x\in\R,\ f(x+T) = f(x).
    $$
    La plus petite valeur de $T$ vérifiant cette propriété est appelée la \highlight{période} de $f$. On appelle \highlight{fréquence} $f_0 = 1/T$ (en Hz) et \highlight{pulsation} $\omega_0 = 2\pi f_0 = 2\pi/T$ (en rad/s) les grandeurs associées.
\end{definition}

\begin{proposition}
    Soit $T>0$ et $f$ une fonction $T$-périodique, alors pour tout $n\in \N$, $nT$ est aussi une période de $f$,
    $$
      \forall x\in\R,\  f(x + nT) = f(x)
    $$
\end{proposition}

\begin{proposition}[Somme de fonctions périodiques]
    Soit $f$ une fonction $T$-périodique et $g$ une fonction $T^\prime$-périodique, avec $T$ et $T^\prime$ tels qu'il existe $p,q\in\N^*$ de sorte que
    $$
        pT = qT^\prime.
    $$
    Alors $f+g$ est périodique de période le plus petit commun multiple de $T$ et $T^\prime$.
\end{proposition}

\begin{remark}
    Si le rapport $T/T^\prime$ est irrationnel, $f+g$ n'est en général pas périodique.
\end{remark}

\begin{proposition}[Formules trigonométriques]

    $$
    \begin{aligned}
        \cos(a+b) &= \cos(a)\cos(b) - \sin(a)\sin(b)\\
        \sin(a+b) &= \cos(a)\sin(b) + \cos(b)\sin(a)\\
        \cos^2(t) + \sin^2(t) &= 1
    \end{aligned}
    $$

\end{proposition}

\begin{proposition}[Transformation somme-produit]
    Pour tous réels $p,q$,
    $$
    \begin{aligned}
        \cos(p) + \cos(q) &= 2\cos\!\left(\frac{p+q}{2}\right)\cos\!\left(\frac{p-q}{2}\right) \\
        \sin(p) + \sin(q) &= 2\sin\!\left(\frac{p+q}{2}\right)\cos\!\left(\frac{p-q}{2}\right)
    \end{aligned}
    $$
\end{proposition}

\begin{remark}
    Ces formules s'obtiennent en sommant les formules d'addition de $\cos(a+b)$ et $\cos(a-b)$ (resp. $\sin(a+b)$ et $\sin(a-b)$) avec $a=\frac{p+q}{2}$, $b=\frac{p-q}{2}$.
\end{remark}

\begin{proposition}[Formules d'Euler]
    Pour tout réel $\theta$,
    $$
        e^{\complexj\theta} = \cos(\theta) + \complexj\sin(\theta),
        \qquad
        \cos(\theta) = \frac{e^{\complexj\theta} + e^{-\complexj\theta}}{2},
        \qquad
        \sin(\theta) = \frac{e^{\complexj\theta} - e^{-\complexj\theta}}{2\complexj}.
    $$
\end{proposition}

\begin{proposition}[Exponentielle complexe]
    Pour tous réels $\theta_1,\theta_2$ et tout $n\in\Z$,
    $$
        e^{\complexj\theta_1}\,e^{\complexj\theta_2} = e^{\complexj(\theta_1+\theta_2)},
        \qquad
        \left(e^{\complexj\theta_1}\right)^n = e^{\complexj n\theta_1},
        \qquad
        \overline{e^{\complexj\theta_1}} = e^{-\complexj\theta_1},
        \qquad
        \left|e^{\complexj\theta_1}\right| = 1.
    $$
    $$
        e^{-\complexj\theta} = e^{\complexj \theta \times (-1)} = (e^{
            \complexj\theta})^{-1} = \frac{1}{e^{\complexj\theta}}
    $$
\end{proposition}

\begin{proposition}
    Les fonctions $\cos$, $\sin$ et $e^{\complexj t}$ sont dérivables et
    $$
        \begin{aligned}
            \cos^\prime(t) &= -\sin(t) \\
            \sin^\prime(t) &= \cos(t) \\
            (e^{\complexj t})^\prime &= \complexj e^{\complexj t} =e^{\complexj (t+\frac{\pi}{2})} 
        \end{aligned}
    $$
\end{proposition}

\section{Série de Fourier}

\subsection{Séries}

Une \highlight{série} est une généralisation de la notion de somme à une succession infinie de termes. Étant donnée une suite $(u_n)_{n\in\N}$, on appelle \highlight{somme partielle} d'ordre $N$ la quantité
$$
    S_N = \sum_{n=0}^{N} u_n = u_0 + u_1 + \dots + u_N.
$$
On dit que la série $\sum u_n$ \highlight{converge} vers une limite $S$ si la suite $(S_N)_{N\in\N}$ converge vers $S$. On note alors
$$
    \sum_{n=0}^{+\infty} u_n = \lim_{N\to+\infty} S_N = S.
$$
Si $(S_N)$ ne converge pas, on dit que la série \highlight{diverge} : la somme infinie n'a alors pas de valeur finie.

\begin{remark}
    Toutes les séries ne convergent pas. Ce n'est pas parce qu'une somme comporte une infinité de termes, chacun de plus en plus petit, qu'elle converge nécessairement vers une valeur finie.
\end{remark}

\begin{exemple}[Série harmonique --- exemple de divergence]
    La série de terme général $u_n = \dfrac{1}{n}$ diverge :
    $$
        \sum_{n=1}^{+\infty} \frac{1}{n} = 1 + \frac12 + \frac13 + \cdots = +\infty,
    $$
    bien que $u_n \to 0$ quand $n\to+\infty$. La convergence vers $0$ du terme général est donc nécessaire, mais pas suffisante, pour que la série converge.
\end{exemple}

\begin{exemple}[Une série convergente]
    La série de terme général $u_n = \dfrac{1}{n^2}$ converge, et on peut montrer que
    $$
        \sum_{n=1}^{+\infty} \frac{1}{n^2} = \frac{\pi^2}{6}.
    $$
\end{exemple}

\begin{definition}[Série trigonométrique]
    Une série trigonométrique de pulsation $\omega_0$ est une série de fonctions
    $$
        a_0 + \sum_{n=1}^{+\infty} \left(a_n\cos(n\omega_0 t) + b_n\sin(n\omega_0 t)\right).
    $$
    Pour $t\in\R$ fixé, c'est une série numérique au sens de la définition précédente : elle converge, ou diverge, selon que la suite de ses sommes partielles $S_N(t)$ admet une limite finie quand $N\to+\infty$.
\end{definition}

\begin{remark}
    Rien ne garantit a priori qu'une série trigonométrique converge, pour un $t$ donné. Même lorsqu'elle converge pour tout $t$, rien ne garantit que la fonction obtenue coïncide avec la fonction $f$ dont les coefficients $a_n,b_n$ sont issus. Les conditions de Dirichlet, introduites ci-dessous, répondent à ces deux questions.
\end{remark}


\subsection{Forme trigonométrique}
% (fin de la séance 1 / CM1)

Joseph Fourier a établi, en 1807, que sous certaines conditions, une fonction périodique peut s'écrire comme une somme infinie de sinusoïdes.

\begin{definition}[Conditions de Dirichlet]
    Soit $f$ une fonction $T$-périodique. On dit que $f$ vérifie les conditions de Dirichlet si, sur une période :
    \begin{itemize}
        \item $f$ a un nombre fini de discontinuités, toutes de première espèce (saut fini) ;
        \item $f$ a un nombre fini d'extrema (maxima et minima) ;
        \item $f$ est bornée.
    \end{itemize}
\end{definition}

\begin{theorem}[Développement en série de Fourier --- forme réelle]
    Si $f$ est $T$-périodique et vérifie les conditions de Dirichlet, alors
    $$
        f(t) = a_0 + \sum_{n=1}^{+\infty} \left(a_n\cos(n\omega_0 t) + b_n\sin(n\omega_0 t)\right), \qquad \omega_0 = \frac{2\pi}{T},
    $$
    avec
    $$
        a_0 = \frac1T\int_0^T f(t)\,\d t, \qquad
        a_n = \frac2T\int_0^T f(t)\cos(n\omega_0 t)\,\d t, \qquad
        b_n = \frac2T\int_0^T f(t)\sin(n\omega_0 t)\,\d t.
    $$
    % En un point de discontinuité, la série converge vers la moyenne des limites à gauche et à droite de $f$.
\end{theorem}

\begin{remark}
    Le coefficient $a_0$ est la valeur moyenne de $f$ sur une période, c'est ce que mesurerait un voltmètre en position DC.
\end{remark}

\begin{proposition}[Linéarité]
    Si $f$ et $g$ sont $T$-périodiques, de coefficients de Fourier respectifs $(a_n,b_n)$ et $(a'_n,b'_n)$, alors pour tous réels $\lambda,\mu$, la fonction $\lambda f+\mu g$ est $T$-périodique.
    
    Ses coefficients de Fourier sont $(\lambda a_n+\mu a'_n,\ \lambda b_n+\mu b'_n)$ (conséquence directe de la linéarité de l'intégrale, Propriété~1.2).
\end{proposition}

\begin{corollary}[Parité]
    Soit $f$ une fonction $T$-périodique vérifiant les conditions de Dirichlet.
    \begin{itemize}
        \item
        Si $f$ est paire, alors $b_n=0$ pour tout $n$ : sa série de Fourier est une série de cosinus.
        \item
        Si $f$ est impaire, alors $a_0=0$ et $a_n=0$ pour tout $n$ : sa série de Fourier est une série de sinus.
    \end{itemize}
\end{corollary}

% \begin{proof}
%     L'intégrale d'une fonction périodique ne dépend pas du choix de l'intervalle d'intégration, pourvu qu'il soit de longueur $T$ : on peut donc calculer $a_n$ et $b_n$ sur $[-\frac T2,\frac T2]$ plutôt que sur $[0,T]$. Le produit d'une fonction paire par $\cos$ est pair, par $\sin$ est impair (et réciproquement si $f$ est impaire) ; on conclut avec la Propriété~1.3.
% \end{proof}

\begin{exemple}[Coefficients du signal créneau]
    Pour $f_{\rm carr\acute{e}}$, la valeur moyenne est nulle par symétrie (autant de temps à $+1$ qu'à $-1$), donc $a_0=0$. La fonction est impaire, donc $a_n=0$ pour tout $n$ (Corollaire~3.1). Il reste
    $$
        b_n = \frac2T\int_0^T f_{\rm carr\acute{e}}(t)\sin(n\omega_0 t)\,\d t = \frac{2\left(1-(-1)^n\right)}{n\pi} = \begin{cases} \dfrac{4}{n\pi} & n \text{ impair} \\ 0 & n \text{ pair.} \end{cases}
    $$
    Jusqu'à l'ordre 5 : $b_1 = \frac{4}{\pi}$, $b_3 = \frac{4}{3\pi}$, $b_5 = \frac{4}{5\pi}$.
\end{exemple}

Les coefficients $a_n$ et $b_n$ mélangent, pour chaque harmonique, une information d'amplitude et une information de phase, ce qui les rend peu lisibles isolément. On préfère souvent une seule grandeur par harmonique, montrant directement quelles fréquences composent le signal et avec quel poids.

\begin{definition}[Spectre d'amplitude]
    Pour $n\geq1$, on pose $A_n = \sqrt{a_n^2+b_n^2}$, de sorte que $a_n\cos(n\omega_0 t) + b_n\sin(n\omega_0 t)$ s'écrit $A_n\cos(n\omega_0 t - \varphi_n)$ pour une certaine phase $\varphi_n$. Le \highlight{spectre d'amplitude} de $f$ est le diagramme en bâtons représentant $A_n$ en fonction de la fréquence $nf_0$ : un trait vertical de hauteur $A_n$ à chaque harmonique $n\geq1$, et rien entre deux harmoniques.
\end{definition}

\subsection{Forme complexe}

La forme réelle demande de manipuler trois familles de coefficients $a_0, a_n, b_n$. Les formules d'Euler permettent de n'en garder qu'une seule.

\begin{theorem}[Développement en série de Fourier --- forme complexe]
    Sous les mêmes conditions que le Théorème~3.1, on peut écrire toute fonction $T$-périodique $f$,
    $$
        f(t) = \sum_{n=-\infty}^{+\infty} c_n\, e^{\complexj n\omega_0 t}, \qquad c_n = \frac1T\int_0^T f(t)\,e^{-\complexj n\omega_0 t}\,\d t.
    $$
\end{theorem}

\begin{proof}
    En remplaçant $\cos(n\omega_0 t)$ et $\sin(n\omega_0 t)$ par leur expression en exponentielles complexes (Propriété~2.6) dans la forme réelle, puis en regroupant les termes en $e^{\complexj n\omega_0 t}$, on obtient $c_0 = a_0$ et, pour $n\geq1$,
    $$
        c_n = \frac{a_n - \complexj b_n}{2}, \qquad c_{-n} = \overline{c_n}.
    $$
\end{proof}

\begin{remark}
    Pour $n\geq1$, $|c_n| = |c_{-n}| = \frac{A_n}{2}$ : l'amplitude de chaque harmonique se répartit pour moitié sur la fréquence $nf_0$ et pour moitié sur $-nf_0$.
\end{remark}

\begin{exemple}[Coefficients complexes du signal créneau]
    Pour $f_{\rm carr\acute{e}}$, $a_n=0$ donc $c_n = -\complexj b_n/2$. Pour $n$ impair, $c_n = -\dfrac{2\complexj}{n\pi}$ et $c_{-n} = \dfrac{2\complexj}{n\pi}$ ; pour $n$ pair (et $n=0$), $c_n=0$.
\end{exemple}

Le spectre d'amplitude défini plus haut n'est indexé que par $n\geq1$. La forme complexe en donne naturellement une version indexée par $n\in\Z$, qui inclut les fréquences négatives introduites par l'écriture en exponentielles.

\begin{definition}[Spectre d'amplitude bilatéral]
    Le \highlight{spectre d'amplitude bilatéral} de $f$ est le diagramme en bâtons représentant $|c_n|$ en fonction de la fréquence $nf_0$, pour $n\in\Z$.
\end{definition}

    D'après l'Exemple~3.5, $|c_n|=2/(n\pi)$ pour $n$ impair (et son symétrique en $-n$), $0$ sinon. On retrouve $|c_1|=|c_{-1}|\approx0{,}64$, soit la moitié de $A_1$ (Remarque~3.4).
\end{exemple}

\begin{remark}
    Les fréquences négatives n'ont pas de sens physique : elles proviennent uniquement de l'écriture complexe, qui combine des termes en $e^{\complexj n\omega_0 t}$ et $e^{-\complexj n\omega_0 t}$ pour reconstituer un signal réel.
\end{remark}

\begin{theorem}[Égalité de Parseval]
    $$
        \frac1T\int_0^T |f(t)|^2\,\d t = \sum_{n=-\infty}^{+\infty} |c_n|^2 = a_0^2 + \frac12\sum_{n=1}^{+\infty}\left(a_n^2+b_n^2\right).
    $$
\end{theorem}

\begin{proof}
    En écrivant $f(t) = \sum_n c_n e^{\complexj n\omega_0 t}$, on a $|f(t)|^2 = \sum_{n,m} c_n\overline{c_m}\,e^{\complexj(n-m)\omega_0 t}$. Or $\frac1T\int_0^T e^{\complexj(n-m)\omega_0 t}\,\d t$ vaut $1$ si $n=m$ et $0$ sinon. Les termes croisés ($n\neq m$) s'annulent donc à l'intégration, et il reste $\frac1T\int_0^T |f(t)|^2\,\d t = \sum_n |c_n|^2$.
\end{proof}
