\section{Fonction d'une variable discrète à valeurs réelles}

Ce chapitre introduit les signaux dont la variable est discrète, 
indexée par un entier plutôt que repérée par un réel. Il introduit l'outil d'analyse spectrale qui
leur correspond, la transformée de Fourier discrète. L'objet central
n'est plus une fonction du temps continu, mais une suite de valeurs.

\begin{definition}[Signal discret]
    Un \highlight{signal discret} est une fonction $x:\Z\to\R$,
    $n\mapsto x[n]$. L'entier $n$ est l'indice (ou le rang) de
    l'échantillon $x[n]$.
\end{definition}

\begin{remark}
    Un signal discret peut provenir des échantillons d'un signal
    continu, $x[n] = x_c(nT_e)$ (avec $T_e$ la période d'échantillonnage),
    mais ce n'est pas la seule origine possible.
    
    Une suite de mesures, un compteur, ou tout processus indexé par un entier définit tout
    aussi un signal discret. Dans ce chapitre, $x[n]$ est étudié
    pour lui-même, indépendamment de son origine.
\end{remark}

\begin{definition}[Échelon unité discret]
    $$
        u[n] = \begin{cases} 1 & \text{si } n\geq0 \\ 0 & \text{si } n<0. \end{cases}
    $$
\end{definition}

\begin{definition}[Impulsion unité discrète]
    $$
        \delta[n] = \begin{cases} 1 & \text{si } n=0 \\ 0 & \text{si } n\neq0. \end{cases}
    $$
\end{definition}

\begin{remark}
    $\delta[n]$ n'est pas l'impulsion de Dirac $\delta(t)$ du chapitre
    précédent. $\delta(t)$ est une distribution, définie uniquement par
    son action sous une intégrale ($\int x(t)\delta(t)\,\d t = x(0)$) :
    elle n'a pas de valeur ponctuelle propre. $\delta[n]$, au contraire,
    est une fonction ordinaire, parfaitement définie point par point.
    Les crochets rappellent cette différence de nature à chaque
    écriture.
\end{remark}

\begin{definition}[Signal périodique discret]
    $x[n]$ est \highlight{périodique} de période $N\in\N^*$ si
    $$x[n+N]=x[n], \ \text{pour tout} \ n\in\Z.$$ Contrairement au cas continu, où
    la période $T$ peut être n'importe quel réel strictement positif,
    la période $N$ d'un signal discret est nécessairement un entier.
\end{definition}

\section{Discret en temps \texorpdfstring{$\Leftrightarrow$}{<=>} périodique en fréquence}

On établit ici le lien entre les fonctions de variable \highlight{discrète} et les fonctions \highlight{périodiques},
à travers la transformée de Fourier.  
Ce lien permet de définit une transformée de Fourier discrète et donc une analyse spectrale sur signaux discret.

\begin{definition}[Peigne de Dirac]
    Pour $T_e>0$, le \highlight{peigne de Dirac} de période $T_e$ est une suite d'impulsion de Dirac, 
    $$
       t\in \R, \ \Sha_{T_e}(t) = \sum_{n=-\infty}^{+\infty} \delta(t-nT_e).
    $$
    Un signal discret $x[n]$ correspond, en temps continu, à un signal
    $x_c(t)\times \Sha_{T_e}(t)$, c'est à dire nul partout sauf aux instants $nT_e$,
            $$x_c(t)= \begin{cases}
                0 & t\neq nT_e, \ n\in \Z\\
                x[n] & t = nT_e, \ n\in\Z
            \end{cases}$$.
\end{definition}

\begin{proposition}[Transformée de Fourier du peigne de Dirac]
    \label{prop:tfpeigne}
    $$
        \Sha_{T_e}(t) \;\tf\; F_e\,\Sha_{F_e}(f), \qquad F_e = 1/T_e.
    $$
\end{proposition}
\begin{proof}
    $\Sha_{T_e}$ est $T_e$-périodique et ne contient, sur une période,
    qu'une seule impulsion (en $t=0$). Ses coefficients de Fourier
    complexes valent donc tous
    $$
        c_n = \frac1{T_e}\int_{-T_e/2}^{T_e/2} \Sha_{T_e}(t)\,e^{-2\complexj\pi nF_et}\,\d t
            = \frac1{T_e}\int_{-T_e/2}^{T_e/2} \delta(t)\,e^{-2\complexj\pi nF_et}\,\d t
            = \frac1{T_e},
    $$
    par propriété d'échantillonnage de $\delta$. La série de Fourier
    correspondante, $\Sha_{T_e}(t) = \frac1{T_e}\sum_n e^{2\complexj\pi nF_et}$,
    se transforme terme à terme : chaque exponentielle a pour
    transformée une impulsion, $\mathcal F[e^{2\complexj\pi nF_et}](f) =
    \delta(f-nF_e)$ (chapitre précédent). Par linéarité,
    $$
        \mathcal F[\Sha_{T_e}](f) = \frac1{T_e}\sum_{n=-\infty}^{+\infty} \delta(f-nF_e) = F_e\,\Sha_{F_e}(f).
    $$
\end{proof}

\begin{theorem}[Dualité discret $\Leftrightarrow$ périodique]
    \label{thm:dualite}
    Soit $x$ un signal continu, de transformée de Fourier $X$.
    \begin{enumerate}[a)]
        \item Si $x$ est rendu discret par $x_e(t)=x(t)\,\Sha_{T_e}(t)$,
        alors
        $$
            X_e(f) = F_e \sum_{k=-\infty}^{+\infty} X(f-kF_e),
        $$
        $X_e$ est \highlight{périodique}, de période $F_e$.
        \item Réciproquement, si $x$ est \highlight{périodique} de
        période $T$, de coefficients de Fourier complexes $(c_n)$, alors
        $$
            X(f) = \sum_{n=-\infty}^{+\infty} c_n\,\delta(f-n/T),
        $$
        $X$ est un peigne de Dirac pondéré, d'espacement $1/T$, c'est une fonction discrète.
    \end{enumerate}
    Dans les deux cas, la transformée de Fourier échange les rôles de
    discret et de périodique entre les deux domaines.
\end{theorem}

\begin{remark}
    On considère une fonction discrète écrite en temps continu et issue de l'échantillonnage d'une fonction $x(t)$
    $$
        x_e(t) =  x(t)\,\Sha_{T_e}(t) = \sum_n
x[n]\,\delta(t-nT_e),
    $$
    avec $x[n] = x_e(nT_e)$. 
    Sa transformée de Fourier peut alors s'écrire
    $$
    X_e(f) = \sum_{n=-\infty}^{+\infty} x[n]\,e^{-2\complexj\pi fnT_e}.
    $$
    Cette expression ne dépend de $f$ et $T_e$ qu'à travers leur produit, 
    la variable pertinente est alors le rapport $\nu=f/F_e=fT_e$ et $X_e(\nu)$ est 1-périodique.
\end{remark}


\begin{definition}[Fréquence normalisée]
    On appelle \highlight{fréquence normalisée} la quantité sans
    dimension $\nu = f/F_e$. Pour une fonction discrète $x[n]$, on écrit sa transformée de Fourier en fonction de $\nu$,
    $$
        X(\nu) = \sum_{n=-\infty}^{+\infty} x[n]\,e^{-2\complexj\pi\nu n},
    $$
    périodique de période $1$ (conséquence directe de la
    $F_e$-périodicité de $X_e(f)$).
\end{definition}

\section{Transformée de Fourier Discrète}

On dispose maintenant de tous les ingrédients pour construire l'outil
central de ce chapitre : un signal discret de longueur finie a un
spectre qui tient dans un nombre fini de valeurs.

\begin{corollary}[Spectre d'une séquence finie]
    Soit une séquence de longueur $N$, $$x[n], \quad  n=0,\dots,N-1,$$ 
    prolongée en un signal $N$-périodique : 
    $$\text{pour} \ k\in\Z, n\in[0,N-1], x[n + kN] = x[n].$$
    
    Ainsi, $x$ est discret en $n$ et $N$-périodique donc d'après le théorème~\ref{thm:dualite}, 
    $X(\nu)$ est aussi discret et 1-périodique, d'espacement $1/N$.
    \textbf{Le spectre d'une séquence de longueur $N$ est donc périodique et tient donc dans exactement
    $N$ valeurs, aux fréquences normalisées $\nu=k/N$, $k=0,\dots,N-1$.}
\end{corollary}

\begin{definition}[Transformée de Fourier Discrète]
    Pour une séquence $x[n]$, $n=0,\dots,N-1$, la
    \highlight{transformée de Fourier discrète} (TFD) est la séquence
    $$
        X[k] = \sum_{n=0}^{N-1} x[n]\,e^{-2\complexj\pi kn/N}, \qquad k=0,\dots,N-1.
    $$
    $X[k]$ est la valeur du spectre normalisé prise en $\nu=k/N$ --
    exactement les $N$ valeurs prédites par le corollaire ci-dessus.
\end{definition}

\begin{definition}[Transformée de Fourier Discrète inverse]
    $$
        x[n] = \frac1N \sum_{k=0}^{N-1} X[k]\,e^{2\complexj\pi kn/N}, \qquad n=0,\dots,N-1.
    $$
\end{definition}

\begin{remark}[Convention de normalisation]
    Le facteur $1/N$ ci-dessus est un choix de convention parmi
    d'autres : certains ouvrages le placent sur la transformée directe
    plutôt que sur l'inverse, ou répartissent $1/\sqrt N$ sur les deux.
    La convention retenue ici place tout le facteur $1/N$ sur la TFD
    inverse.
\end{remark}

\begin{exemple}[TFD d'un cosinus échantillonné]
    Soit $N=8$ et $x[n]=\cos(2\pi\cdot2n/8)=\cos(\pi n/2)$ (deux cycles
    sur la fenêtre). Avec la formule d'Euler,
    $x[n]=\tfrac12e^{2\complexj\pi\cdot2n/8}+\tfrac12e^{-2\complexj\pi\cdot2n/8}$,
    et chaque somme géométrique $\sum_{n=0}^{7}e^{-2\complexj\pi(k\mp2)n/8}$
    vaut $8$ si $k\mp2\equiv0\ (\mathrm{mod}\ 8)$, et $0$ sinon. D'où
    $$
        X[2]=X[6]=4, \qquad X[k]=0 \text{ pour les autres } k.
    $$
    Deux valeurs non nulles, symétriques autour de $k=N/2$ : même
    structure que les deux impulsions de la transformée d'un cosinus
    continu (chapitre précédent).
\end{exemple}

\begin{exemple}[TFD d'une porte discrète]
    Soit $x[n]=1$ pour $n=0,\dots,M-1$ et $x[n]=0$ pour
    $n=M,\dots,N-1$ (une porte discrète de largeur $M$ dans une fenêtre
    de $N$ points). La somme géométrique se calcule directement :
    $$
        X[k] = \sum_{n=0}^{M-1} e^{-2\complexj\pi kn/N}
             = e^{-\complexj\pi k(M-1)/N}\,\frac{\sin(\pi kM/N)}{\sin(\pi k/N)}
             \quad (k\neq0), \qquad X[0]=M.
    $$
    C'est l'analogue discret du sinus cardinal : un lobe principal
    autour de $k=0$, d'autant plus large que $M$ est petit devant $N$ --
    même compromis largeur temporelle / largeur spectrale que pour la
    porte continue du chapitre précédent.
\end{exemple}

\begin{remark}[Algorithme de calcul rapide]
    Calculer $X[k]$ pour tous les $k$ à partir de la définition demande
    $N^2$ multiplications ($N$ valeurs de $k$, chacune une somme de $N$
    termes). En séparant les indices pairs et impairs de $x[n]$, chaque
    TFD de taille $N$ se ramène à deux TFD de taille $N/2$ : répétée
    récursivement, cette idée ramène le coût total à $N\log_2N$
    opérations. C'est l'algorithme de \highlight{transformée de Fourier
    rapide} (FFT), utilisé en pratique dès que $N$ dépasse quelques
    dizaines de points.
\end{remark}

\begin{remark}[Discontinuité du prolongement périodique et fuite spectrale]
    La TFD suppose implicitement que $x[n]$ se prolonge en un signal
    $N$-périodique (corollaire ci-dessus). Si le signal observé ne
    boucle pas exactement sur $N$ échantillons -- par exemple un
    cosinus dont la fréquence n'est pas un multiple entier de $F_e/N$ --,
    ce prolongement présente une discontinuité au raccord
    $n=N-1\to n=0$. Or un signal périodique discontinu a des
    coefficients de Fourier qui décroissent lentement (chapitre 1) : la
    TFD d'un tel signal n'est plus concentrée sur un ou deux rangs $k$,
    mais s'étale sur plusieurs -- un phénomène appelé
    \highlight{fuite spectrale}.
\end{remark}
