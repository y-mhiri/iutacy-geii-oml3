%% Produit x(u) y(a-u) pour t=a=2 (a>0) : porte reflechie/decalee
%% (rouge), rampe x(u) restreinte a la fenetre (bleu), aire coloree =
%% (x*y)(a) = integrale du produit. Parties hors fenetre en pointille
%% clair pour le contexte. Consomme par slides.
\begin{tikzpicture}[x=0.9cm,y=0.6cm]
    \draw[-Latex] (-3.5,0) -- (3.5,0) node[right] {\small $u$};
    \draw[-Latex] (0,-0.5) -- (0,3.5);

    % contexte (hors fenetre), estompe
    \draw[niceblue!35, thick, dashed, domain=-3:1] plot (\x,\x);
    \draw[niceblue!35, thick, dashed, domain=3:3.3] plot (\x,\x);
    \draw[red!35, thick, dashed] (-3,0) -- (1,0);
    \draw[red!35, thick, dashed] (3,0) -- (3.3,0);

    % aire du produit = integrale de a-1 a a+1
    \fill[orange, opacity=0.4] (1,0) -- (1,1) -- (3,3) -- (3,0) -- cycle;

    % la porte reflechie/decalee, centree en a=2
    \draw[red, thick] (1,0) -- (1,1) -- (3,1) -- (3,0);
    \node[red] at (2,1.4) {\small $y(a-u)$};

    % la rampe, dans la fenetre
    \draw[niceblue, thick, domain=1:3] plot (\x,\x);
    \node[niceblue] at (2.7,2.5) {\small $x(u)$};

    \draw[dashed] (2,0) -- (2,2);
    \node[below] at (1,0) {\small $a-1$};
    \node[below] at (2,0) {\small $a$};
    \node[below] at (3,0) {\small $a+1$};
\end{tikzpicture}
