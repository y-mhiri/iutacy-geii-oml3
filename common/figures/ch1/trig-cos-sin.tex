\begin{tikzpicture}[x=0.7cm,y=1cm]
    \draw[-Latex] (-0.3,0) -- (4*pi+0.5,0) node[right] {$t$};
    \draw[-Latex] (0,-1.6) -- (0,1.6);
    \draw[niceblue,thick,domain=0:4*pi,smooth,variable=\t,samples=150]
        plot ({\t},{1.3*sin(\t r)});
    \draw[red,thick,domain=0:4*pi,smooth,variable=\t,samples=150]
        plot ({\t},{1.3*cos(\t r)});
    \node[niceblue] at (1.6,1.48) {$\sin(t)$};
    \node[red] at (2*pi,1.48) {$\cos(t)$};
    \draw[dashed] (2*pi,-1.4) -- (2*pi,1.4);
    \draw[latex-latex] (0,-1.45) -- (2*pi,-1.45) node[midway,below] {$2\pi$};
\end{tikzpicture}
