\begin{tikzpicture}[scale=1.8]
    \def\ang{50}
    \draw[-Latex] (-1.4,0) -- (1.5,0) node[right] {\small $\mathrm{Re}$};
    \draw[-Latex] (0,-1.4) -- (0,1.5) node[above] {\small $\mathrm{Im}$};

    \draw[thick, niceblue] (0,0) -- (\ang:1) coordinate (P);

    \draw[dashed] (P) -- (0,{sin(\ang)});
    \draw (0.3,0) arc (0:\ang:0.3);
    \node at (0.48,0.16) {\small $\theta$};
    \fill (P) circle (0.02);
    \node[above right] at (P) {$e^{\mathsf{j}\theta}$};
