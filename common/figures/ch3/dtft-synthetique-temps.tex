%% Panneau temporel : une séquence discrète finie x[n], n=0..5,
%% valeurs arbitraires (0.6, 1.0, -0.4, 0.8, -0.9, 0.3) -- volontairement
%% "quelconque", pas une forme reconnaissable (porte, etc.), pour
%% montrer que la propriété qui suit (TF périodique mais continue) ne
%% tient pas à une forme particulière. Nulle partout ailleurs (points
%% creux en n=-1 et n=6). Consommé par slides.
\begin{tikzpicture}[x=0.8cm,y=1cm]
    \draw[-{Latex[length=2mm]}] (-2,0) -- (7,0) node[right] {\small$n$};
    \filldraw[shibuidarkgray] (-1,0) circle (1.1pt);
    \foreach \n/\v in {0/0.6, 1/1.0, 2/-0.4, 3/0.8, 4/-0.9, 5/0.3} {
        \draw[shibuidarkgray,ultra thick,-{Latex[length=2mm]}] (\n,0) -- (\n,\v);
    }
    \filldraw[shibuidarkgray] (6,0) circle (1.1pt);
    \draw[red,domain=0.3:2.3,samples=300,smooth]
        plot (-1, \x-1);
    \draw[red,domain=0.3:2.3,samples=300,smooth]
        plot (6, \x-1);
    \node[below left] at (-1,0) {0};
    \node[below right] at (6,0) {N-1};
\end{tikzpicture}
