%% ── Ex 1 : CLASSIFY / DEFINE ────────────────────────────────────
%% Purpose: activate vocabulary from the CM.
%% No computation harder than substituting into a definition.
%% At least one positive, one negative, one boundary case.
\begin{exercice}
    \label{ex:energie-puissance}
    Pour les signaux suivants, préciser s'il s'agit de signaux à énergie finie et/ou puissance finie,
    \begin{enumerate}[\bfseries 1.]
        \item $x(t) = \tan(t)$
        \begin{solution}
            $\tan$ est $\pi$-périodique, mais non bornée : au voisinage de chaque pôle $t=\frac\pi2+k\pi$ ($k\in\Z$), $\tan(t)\sim\frac1{\pi/2-t}$, donc $\tan^2$ n'est déjà pas intégrable sur un intervalle borné contenant un pôle. L'énergie est donc infinie, mais la puissance aussi : contrairement à ce que suggère la remarque du cours (un signal périodique non nul a une puissance finie), cela suppose implicitement que le carré du signal reste intégrable sur une période, ce qui échoue ici à cause des pôles.
        \end{solution}

        \item $x(t) = \cos(2\pi t)$
        \begin{solution}
            $\cos(2\pi t)$ est $1$-périodique et non nulle : son énergie est infinie. Sa puissance vaut
            $$
                P_x = \int_0^1 \cos^2(2\pi t)\,\d t = \int_0^1 \frac{1+\cos(4\pi t)}2\,\d t = \frac12,
            $$
            finie, comme attendu pour un signal périodique borné.
        \end{solution}
        \item $x(t) = \Pi_\theta(t)\, e^{-t/\tau} \sin(2\pi t)$
        \begin{solution}
            $x$ est nulle en dehors de $[-\theta/2,\theta/2]$, et bornée sur cet intervalle (produit de fonctions continues) : c'est un signal transitoire, à énergie finie. Sa puissance moyenne est alors nulle : $P_x=0$ (une énergie finie, divisée par une fenêtre $T$ qui grandit indéfiniment, tend vers $0$).
        \end{solution}
    \end{enumerate}
\end{exercice}


%% ── Ex 2 : PROVE ─────────────────────────────────────────────────
%% Purpose: make students re-derive a theorem stated in the CM.
%% Structure: easy direction → hard direction → verify on an example.
\begin{exercice}
    \label{ex:fenetrage}
    Soit $\theta>0$ et $x:\R\to\R$. On considère le signal fenêtré $g(t) = \Pi_\theta(t)\,x(t)$.
    \begin{enumerate}[\bfseries 1.]
        \item Exprimer $E_g$ comme une intégrale sur un intervalle borné.
        \begin{solution}
            $\Pi_\theta$ ne prend que les valeurs $0$ et $1$, donc $\Pi_\theta(t)^2=\Pi_\theta(t)$, et
            $$
                E_g = \int_{-\infty}^{+\infty} \Pi_\theta(t)^2\,x(t)^2\,\d t = \int_{-\theta/2}^{\theta/2} x(t)^2\,\d t.
            $$
        \end{solution}

        \item En déduire une condition suffisante simple sur $x$ pour que $g$ soit à énergie finie.
        \begin{solution}
            Si $x$ est bornée sur $[-\theta/2,\theta/2]$, par $M$ disons, alors $E_g \le M^2\theta < \infty$ : $g$ est à énergie finie. Plus généralement il suffit que $x$ soit de carré intégrable sur cet intervalle, ce qui est automatique pour toute fonction continue, ou bornée avec un nombre fini de discontinuités.
        \end{solution}

        \item Fenêtrer ne suffit donc pas toujours : à l'aide de l'Exercice~\ref{ex:energie-puissance}, pour quelles valeurs de $\theta$ le signal $g(t) = \Pi_\theta(t)\tan(t)$ est-il à énergie finie ?
        \begin{solution}
            $\tan$ est continue, donc bornée, sur $[-\theta/2,\theta/2]$ si et seulement si cet intervalle ne contient aucun pôle, c'est-à-dire si $\theta/2<\pi/2$, soit $\theta<\pi$ : dans ce cas $g$ est à énergie finie (question précédente). Si $\theta\ge\pi$, la fenêtre contient le pôle $t=\pi/2$ et $E_g$ diverge, exactement comme pour $\tan$ non fenêtrée : fenêtrer temporellement ne rend pas automatiquement un signal à énergie finie si celui-ci garde, à l'intérieur de la fenêtre, une singularité non intégrable.
        \end{solution}
    \end{enumerate}
\end{exercice}


%% ── Ex 3 : APPLY ─────────────────────────────────────────────────
%% Purpose: numerical problem with real or realistic GEII data.
%% Ends with an engineer's conclusion, not just a number.
\begin{exercice}
    \label{ex:support-tf}
    Pour chacune des fonctions suivantes, préciser son support (l'intervalle, ou l'ensemble, sur lequel le signal est non nul) et calculer sa transformée de Fourier, soit par calcul intégral, soit en utilisant les propriétés de la Transformée de Fourier.
    \begin{enumerate}[\bfseries 1.]
        \item $x(t) = \Pi_\theta(t)$
        \begin{solution}
            Support : $]-\theta/2,\theta/2[$. C'est le signal porte vu en cours :
            $$
                \mathcal F[x](f) = \theta\,\sinc(\pi f\theta).
            $$
        \end{solution}

        \item $x(t) = \delta(t-t_0) + \delta(t-t_1)$
        \begin{solution}
            Support : $\{t_0,t_1\}$. Par linéarité et la TF d'une impulsion décalée (vue en cours),
            $$
                \mathcal F[x](f) = e^{-2\complexj\pi ft_0} + e^{-2\complexj\pi ft_1}.
            $$
        \end{solution}
        \item $x(t) = e^{-a|t|}$, avec $a>0$ (ci-dessous, pour $a=2$) :
        \begin{center}
            \begin{tikzpicture}
                \begin{axis}[
                    width=6.5cm, height=4cm,
                    xlabel={$t$}, ylabel={$x(t)$},
                    xmin=-2, xmax=2, ymin=0, ymax=1.3,
                    axis lines=middle,
                    xlabel style={at=(current axis.right of origin), anchor=west},
                    ylabel style={at=(current axis.above origin), anchor=south},
                ]
                \addplot[domain=-2:2, samples=200, niceblue, thick] {exp(-2*abs(x))};
                \end{axis}
            \end{tikzpicture}
        \end{center}
        \begin{solution}
            Support : $\R$ tout entier (contrairement aux deux signaux précédents, $x$ n'est jamais nulle). En séparant l'intégrale en $t<0$ et $t>0$ :
            $$
                \mathcal F[x](f) = \int_{-\infty}^0 e^{at}e^{-2\complexj\pi ft}\,\d t + \int_0^{+\infty} e^{-at}e^{-2\complexj\pi ft}\,\d t = \frac1{a-2\complexj\pi f} + \frac1{a+2\complexj\pi f} = \frac{2a}{a^2+4\pi^2f^2}.
            $$
            Un résultat réel et pair, comme attendu puisque $x$ est réelle et paire (propriété de parité). Plus la décroissance temporelle $a$ est lente, plus le spectre se resserre autour de $f=0$ : on retrouve, une fois de plus, l'antagonisme temps-fréquence -- une courbe en cloche, dite lorentzienne :
            \begin{center}
                \begin{tikzpicture}
                    \begin{axis}[
                        width=6.5cm, height=4cm,
                        xlabel={$f$}, ylabel={$X(f)$},
                        xmin=-1.5, xmax=1.5, ymin=0, ymax=1.3,
                        axis lines=middle,
                        xlabel style={at=(current axis.right of origin), anchor=west},
                        ylabel style={at=(current axis.above origin), anchor=south},
                    ]
                    \addplot[domain=-1.5:1.5, samples=200, niceblue, thick] {1/(1+3.14159^2*x^2)};
                    \end{axis}
                \end{tikzpicture}
            \end{center}
        \end{solution}
    \end{enumerate}
\end{exercice}


\begin{exercice}[Lobes spectraux et compromis temps-fréquence]
    \label{ex:lobes}
    \begin{enumerate}[\bfseries 1.]
        \item Rappeler la transformée de Fourier $X$ de $\Pi_\theta$ (Exercice~\ref{ex:support-tf}). Montrer qu'elle s'annule exactement aux fréquences $f=k/\theta$, $k\in\Z^*$. En déduire la largeur du \highlight{lobe principal} (autour de $0$) puis celle de chaque \highlight{lobe secondaire}.
        \begin{solution}
            $X(f) = \theta\,\sinc(\pi f\theta)$. Comme $\sinc(u)=0 \iff u\in\pi\Z^*$ (et $\sinc(0)=1\neq0$), $X(f)=0 \iff \pi f\theta = k\pi,\ k\in\Z^* \iff f=k/\theta$.

            Le lobe principal s'étend donc entre les deux premiers zéros, $f\in\,]-1/\theta,1/\theta[$ : une largeur $2/\theta$. Chaque lobe secondaire s'étend entre deux zéros consécutifs $f=k/\theta$ et $f=(k+1)/\theta$ : une largeur $1/\theta$. Plus l'impulsion est étroite dans le temps ($\theta$ petit), plus son spectre s'étale en fréquence.
        \end{solution}

        \item Calculer $E_x$, l'énergie de $\Pi_\theta$. En déduire $\displaystyle\int_{-\infty}^{+\infty} |X(f)|^2\,\d f$.
        \begin{solution}
            $\displaystyle E_x = \int_{-\theta/2}^{\theta/2} 1\,\d t = \theta$. Par Parseval-Plancherel, $\displaystyle\int_{-\infty}^{+\infty}|X(f)|^2\,\d f = E_x = \theta$.
        \end{solution}

        \item Calculer, en fonction de $\theta$, l'énergie contenue dans le lobe principal, $\displaystyle I_1 = \int_{-1/\theta}^{1/\theta} |X(f)|^2\,\d f$ (on admettra que $\displaystyle\int_{-\pi}^{\pi}\sinc^2(u)\,\d u \approx 2{,}836$). Quelle fraction de $E_x$ cela représente-t-il ?
        \begin{solution}
            En posant $u=\pi f\theta$ (donc $\d u = \pi\theta\,\d f$, et $f=\pm1/\theta \Rightarrow u=\pm\pi$) :
            $$
                I_1 = \int_{-1/\theta}^{1/\theta} \theta^2\sinc^2(\pi f\theta)\,\d f = \frac{\theta^2}{\pi\theta}\int_{-\pi}^{\pi}\sinc^2(u)\,\d u \approx \frac{2{,}836}\pi\,\theta \approx 0{,}903\,\theta.
            $$
            Soit $I_1/E_x \approx 90{,}3\,\%$ : la quasi-totalité de l'énergie du signal porte est concentrée dans le lobe principal, les lobes secondaires ne portant ensemble que $\approx9{,}7\,\%$.
        \end{solution}

        \item On observe maintenant $y(t) = \Pi_\theta(t)\cos(2\pi f_0t)$, avec $f_0\theta\in\N^*$ (un nombre entier de périodes dans la fenêtre) et $f_0\gg1/\theta$ (de nombreuses périodes dans la fenêtre) -- ci-dessous, pour $\theta=2$, $f_0=4$ : le cosinus (gris) tronqué par la fenêtre (bleu). Calculer $\mathcal F[y]$. Sans nouveau calcul d'intégrale, déduire de la question 3 la fraction de l'énergie de $y$ contenue dans les deux lobes principaux, situés autour de $\pm f_0$.
        \begin{center}
            \begin{tikzpicture}
                \begin{axis}[
                    width=7cm, height=4.3cm,
                    xlabel={$t$},
                    xmin=-3, xmax=3, ymin=-1.3, ymax=1.3,
                    axis lines=middle,
                    xlabel style={at=(current axis.right of origin), anchor=west},
                    ytick=\empty,
                ]
                \addplot[domain=-3:3, samples=600, black!30, thick] {cos(deg(2*3.14159*4*x))};
                \addplot[domain=-1:1, samples=200, niceblue, thick] {cos(deg(2*3.14159*4*x))};
                \draw[dashed] (axis cs:-1,-1.3) -- (axis cs:-1,1.3);
                \draw[dashed] (axis cs:1,-1.3) -- (axis cs:1,1.3);
                \end{axis}
            \end{tikzpicture}
        \end{center}
        \begin{solution}
            Avec la formule d'Euler et le décalage fréquentiel (même méthode que pour la porte) :
            $$
                \mathcal F[y](f) = \frac{\theta}2\Big(\sinc\big(\pi\theta(f-f_0)\big) + \sinc\big(\pi\theta(f+f_0)\big)\Big)
            $$
            (on retrouve l'exemple du cours sur le cosinus fenêtré). Le module de ce spectre : deux lobes de sinus cardinal centrés en $\pm f_0$, avec des lobes secondaires qui subsistent tout le long de l'axe des fréquences.
            \begin{center}
                \begin{tikzpicture}
                    \begin{axis}[
                        width=7cm, height=4.3cm,
                        xlabel={$f$}, ylabel={module},
                        xmin=-7, xmax=7, ymin=0,
                        axis lines=middle,
                        xlabel style={at=(current axis.right of origin), anchor=west},
                        ylabel style={at=(current axis.above origin), anchor=south},
                    ]
                    \addplot[domain=-7:7, samples=400, niceblue, thick]
                        {abs(sin(deg(2*3.14159*(x-4)))/(2*3.14159*(x-4)) + sin(deg(2*3.14159*(x+4)))/(2*3.14159*(x+4)))};
                    \end{axis}
                \end{tikzpicture}
            \end{center}

            L'énergie de $y$ vaut
            $$
                E_y = \int_{-\theta/2}^{\theta/2}\cos^2(2\pi f_0t)\,\d t = \frac\theta2 + \frac{\sin(2\pi f_0\theta)}{4\pi f_0} = \frac\theta2,
            $$
            le second terme s'annulant car $f_0\theta\in\N^*$. Comme $f_0\gg1/\theta$, les deux motifs centrés en $\pm f_0$ ne se recouvrent pas et, par symétrie, portent chacun la moitié de l'énergie, $\theta/4$. Chacun a exactement la forme d'un sinc$^2$ translaté, identique à celle de la porte seule : la fraction $90{,}3\,\%$ de la question 3 s'applique donc à chacun. Au total, $\approx90{,}3\,\%$ de $E_y$ se trouve dans les deux lobes principaux, autour de $\pm f_0$ -- fenêtrer puis moduler ne change pas cette proportion, seulement sa position en fréquence.
        \end{solution}

        \item On considère à présent $z(t) = \sinc(\pi\theta t)\cos(2\pi f_0t)$, avec $f_0>\theta/2$ (pour que les deux motifs du spectre ne se recouvrent pas) -- ci-dessous, pour $\theta=2$, $f_0=4$ : la porteuse (bleu) au sein de l'enveloppe $\pm|\sinc(\pi\theta t)|$ (pointillés), un signal qui ne s'annule identiquement sur aucun intervalle, si loin de l'origine soit-on. En utilisant la propriété de dualité, calculer $\mathcal F[z]$. Quelle fraction de l'énergie de $z$ est contenue dans son lobe principal ?
        \begin{center}
            \begin{tikzpicture}
                \begin{axis}[
                    width=7cm, height=4.3cm,
                    xlabel={$t$},
                    xmin=-3, xmax=3, ymin=-1.3, ymax=1.3,
                    axis lines=middle,
                    xlabel style={at=(current axis.right of origin), anchor=west},
                    ytick=\empty,
                ]
                \addplot[domain=-3:3, samples=200, black!40, dashed] {abs(sin(deg(2*3.14159*x))/(2*3.14159*x))};
                \addplot[domain=-3:3, samples=200, black!40, dashed] {-abs(sin(deg(2*3.14159*x))/(2*3.14159*x))};
                \addplot[domain=-3:3, samples=600, niceblue, thick] {sin(deg(2*3.14159*x))/(2*3.14159*x)*cos(deg(2*3.14159*4*x))};
                \end{axis}
            \end{tikzpicture}
        \end{center}
        \begin{solution}
            La dualité, appliquée à $\Pi_\theta(t) \overset{\mathcal F}{\longleftrightarrow} \theta\sinc(\pi\theta f)$, donne $\mathcal F[\theta\sinc(\pi\theta\cdot)](f) = \Pi_\theta(-f) = \Pi_\theta(f)$ ($\Pi_\theta$ étant paire), soit
            $$
                \sinc(\pi\theta t) \overset{\mathcal F}{\longleftrightarrow} \frac1\theta\,\Pi_\theta(f).
            $$
            Par le même argument que pour $y$ (Euler puis décalage fréquentiel) :
            $$
                \mathcal F[z](f) = \frac1{2\theta}\Big(\Pi_\theta(f-f_0) + \Pi_\theta(f+f_0)\Big).
            $$
            Le module de ce spectre : deux portes exactes, sans aucun lobe secondaire.
            \begin{center}
                \begin{tikzpicture}
                    \begin{axis}[
                        width=7cm, height=4.3cm,
                        xlabel={$f$}, ylabel={module},
                        xmin=-7, xmax=7, ymin=0, ymax=0.35,
                        axis lines=middle,
                        xlabel style={at=(current axis.right of origin), anchor=west},
                        ylabel style={at=(current axis.above origin), anchor=south},
                    ]
                    \draw[niceblue,thick] (axis cs:3,0) -- (axis cs:3,0.25) -- (axis cs:5,0.25) -- (axis cs:5,0);
                    \draw[niceblue,thick] (axis cs:-5,0) -- (axis cs:-5,0.25) -- (axis cs:-3,0.25) -- (axis cs:-3,0);
                    \end{axis}
                \end{tikzpicture}
            \end{center}

            Cette fois le spectre est \emph{nul} en dehors de deux bandes de largeur $\theta$ centrées en $\pm f_0$ : il n'y a aucun lobe secondaire, toute l'énergie de $z$ est contenue dans ces deux bandes, soit $100\,\%$ -- contre $90{,}3\,\%$ pour $y$.
        \end{solution}

        \item Conclure, en une ou deux phrases, sur ce que ces deux exemples disent de la relation entre la durée d'un signal dans le temps et la largeur de son spectre.
        \begin{solution}
            $\Pi_\theta\cos$, de durée bornée dans le temps, a un spectre qui s'étend en réalité sur toutes les fréquences : des lobes secondaires subsistent indéfiniment, même s'ils ne portent que peu d'énergie. À l'inverse, $\sinc(\pi\theta\cdot)\cos$, dont le spectre est borné en fréquence, doit nécessairement s'étendre sur tout l'axe des temps ($\sinc$ ne s'annule identiquement sur aucun intervalle). Un signal ne peut donc pas être simultanément limité en durée \emph{et} en bande : gagner en concentration temporelle se paie en étalement spectral, et réciproquement.
        \end{solution}
    \end{enumerate}
\end{exercice}
