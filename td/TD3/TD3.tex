\begin{exercice}
    Soit $x(t)$ une fonction de $\R$ à valeurs réelles. On notera $X(f)$ sa transformée de Fourier.
    \begin{enumerate}[\bfseries 1.]
        \item Rappeler la formule de la transformée de Fourier et de son inverse.
        \begin{solution}
            $$
                X(f) = \mathcal F[x](f) = \int_{-\infty}^{+\infty} x(t)\,e^{-2\complexj\pi ft}\,\d t,
                \qquad
                x(t) = \mathcal F^{-1}[X](t) = \int_{-\infty}^{+\infty} X(f)\,e^{2\complexj\pi ft}\,\d f.
            $$
        \end{solution}
        \item Que représente la variable $f$ de la transformée de Fourier ?
        \begin{solution}
            Une fréquence, en $\Hz$.
        \end{solution}
        \item Dans quel espace les valeurs de $X(f)$ vivent-elles ?
        \begin{solution}
            Dans $\C$ : même si $x$ est à valeurs réelles, $X(f)$ est a priori complexe (l'exponentielle $e^{-2\complexj\pi ft}$ l'est).
        \end{solution}
        \item Quelle formule du cours traduit l'idée que la transformée de Fourier effectue une bijection entre le domaine fréquentiel et le domaine temporel ?
        \begin{solution}
            $\mathcal F^{-1}[\mathcal F[x]] = x$.
        \end{solution}
    \end{enumerate}
\end{exercice}

\begin{exercice}
    On considère un système dont la sortie s'écrit
    $$
        y(t) = \begin{cases}
        e^{-\alpha t} & t \geq 0 \\
        0 & t < 0,
        \end{cases}
    $$ avec $\alpha > 0$.
    \begin{enumerate}[\bfseries 1.]
        \item Calculer $Y(f)$, la transformée de Fourier de $y(t)$.

        \textit{On remarquera que $y$ est nulle pour $t < 0$~; l'intégrale ne
        porte donc que sur $[0, +\infty[$.}
        \begin{solution}
            $$
                Y(f) = \int_0^{+\infty} e^{-\alpha t}e^{-2\complexj\pi ft}\,\d t
                = \int_0^{+\infty} e^{-(\alpha+2\complexj\pi f)t}\,\d t
                = \left[\frac{e^{-(\alpha+2\complexj\pi f)t}}{-(\alpha+2\complexj\pi f)}\right]_0^{+\infty}
                = \frac{1}{\alpha+2\complexj\pi f},
            $$
            le terme en $+\infty$ s'annulant car $\alpha>0$ (partie réelle de l'exposant strictement négative).
        \end{solution}

        \item Vérifier que
        $$\abs{Y(f)}^2 = \frac{1}{\alpha^2 + 4\pi^2 f^2}
                       = \frac{1/\alpha^2}{1 + \omega^2/\alpha^2},
          \qquad \omega = 2\pi f.$$
        \begin{solution}
            $$
                \abs{Y(f)}^2 = Y(f)\overline{Y(f)} = \frac{1}{\alpha+2\complexj\pi f}\cdot\frac{1}{\alpha-2\complexj\pi f} = \frac{1}{\alpha^2+4\pi^2f^2}.
            $$
            En factorisant $\alpha^2$ au dénominateur, $\dfrac{1}{\alpha^2+4\pi^2f^2} = \dfrac{1/\alpha^2}{1+4\pi^2f^2/\alpha^2} = \dfrac{1/\alpha^2}{1+\omega^2/\alpha^2}$.
        \end{solution}

        \item On y reconnaît le carré du gain d'un filtre RC passe-bas.
        Donner la fréquence de coupure $f_c$ en fonction de $\alpha$. À l'aide du comportement de
        $\abs{Y(f)}$ pour $f \gg f_c$, justifier que les hautes fréquences de
        l'entrée sont atténuées en sortie.
        \begin{solution}
            La forme $\dfrac{1/\alpha^2}{1+(\omega/\alpha)^2}$ s'identifie à $\dfrac{H_0^2}{1+(\omega/\omega_c)^2}$ avec $\omega_c=\alpha=2\pi f_c$, soit
            $$
                f_c = \frac{\alpha}{2\pi}.
            $$
            En interprétant $y(t)$ comme la réponse impulsionnelle du système (sa sortie pour une entrée $\delta(t)$, cf. Exercice~3), $\abs{Y(f)}$ en est le gain. Pour $f\gg f_c$, $\omega^2/\alpha^2\gg1$ donc
            $$
                \abs{Y(f)}^2 \approx \frac{1/\alpha^2}{\omega^2/\alpha^2} = \frac{1}{4\pi^2f^2} \xrightarrow[f\to+\infty]{} 0 :
            $$
            le gain décroît avec $f$, les hautes fréquences de l'entrée sont donc atténuées en sortie.
        \end{solution}
    \end{enumerate}
\end{exercice}

\begin{exercice}
    Pour rappel, une impulsion de Dirac peut se définir comme la limite d'une fonction porte centrée en 0 et \highlight{d'aire 1} quand la largeur de la porte tend vers 0. On la note $\delta(t)$ de sorte que, quelle que soit la fonction $x(t)$ choisie,
    $$
        \delta(t)\,x(t) = x(0)\,\delta(t).
    $$

    \begin{enumerate}[\bfseries 1.]
        \item À partir de la définition de l'impulsion de Dirac, donner la valeur de $$I_\delta = \int_{-\infty}^{+\infty} \delta(t) \d t.$$
        \begin{solution}
            $I_\delta = 1$ : c'est l'aire de la porte limite, par construction.
        \end{solution}
        \item Soit $x(t)$ une fonction à valeurs réelles. En déduire la valeur de $$\int_{-\infty}^{+\infty} x(t)\delta(t)\d t.$$
        \begin{solution}
            D'après l'identité rappelée en tête d'exercice, $x(t)\delta(t)=x(0)\delta(t)$, donc
            $$
                \int_{-\infty}^{+\infty} x(t)\delta(t)\,\d t = x(0)\int_{-\infty}^{+\infty}\delta(t)\,\d t = x(0)\times I_\delta = x(0).
            $$
        \end{solution}
        \item Soit $t_0 \in\R$. Que vaut
         $$\int_{-\infty}^{+\infty} x(t)\delta(t-t_0)\d t\ ?$$
        \begin{solution}
            En posant $u=t-t_0$ (donc $t=u+t_0$, $\d u=\d t$), on se ramène à la question précédente appliquée à la fonction $u\mapsto x(u+t_0)$ :
            $$
                \int_{-\infty}^{+\infty} x(t)\delta(t-t_0)\,\d t = \int_{-\infty}^{+\infty} x(u+t_0)\delta(u)\,\d u = x(0+t_0) = x(t_0).
            $$
        \end{solution}
        \item Rappeler la définition du produit de convolution puis justifier à partir du résultat de la question 3 que l'impulsion de Dirac est l'élément neutre de l'opération de convolution.

        \textit{Rappel :  un élément neutre laisse tout élément inchangé par l'opération. Ainsi $0$ pour l'addition ($a+0=a$) et $1$ pour la multiplication ($a\times 1=a$). Ici : $x * \delta = x$ pour toute fonction $x$.}
        \begin{solution}
            $$
                (x*y)(t) = \int_{-\infty}^{+\infty} x(\tau)\,y(t-\tau)\,\d\tau.
            $$
            Donc
            $$
                (x*\delta)(t) = \int_{-\infty}^{+\infty} x(\tau)\,\delta(t-\tau)\,\d\tau.
            $$
            On pose $u=t-\tau$ (donc $\tau=t-u$, $\d\tau=-\d u$, et les bornes s'inversent, ce qui compense le signe) :
            $$
                (x*\delta)(t) = \int_{-\infty}^{+\infty} x(t-u)\,\delta(u)\,\d u.
            $$
            En appliquant le résultat de la question 2 à la fonction $u\mapsto x(t-u)$ (prise en $u=0$) :
            $$
                (x*\delta)(t) = x(t-0) = x(t).
            $$
            Donc $x*\delta=x$ pour toute fonction $x$ : $\delta$ est bien l'élément neutre de la convolution.
        \end{solution}
    \end{enumerate}

\end{exercice}

\begin{exercice}
    On considère la fonction $x(t) = \cos(\omega t)$.

    \begin{enumerate}[\bfseries 1.]
        \item Sur quel intervalle la fonction $x(t)$ est-elle définie ?
        \begin{solution}
            $\R$ tout entier : le cosinus est défini pour tout réel.
        \end{solution}
        \item On souhaite modéliser un cosinus observé sur un temps fini. On l'exprimera par une fonction $y(t) = k(t)x(t)$. Définir la fonction $k(t)$ telle que
        $$
            y(t) = k(t)x(t) = \begin{cases}
                x(t) & t\in[-\theta/2, \theta/2] \\
                0 & \text{sinon}
            \end{cases}
        $$
        À quelle fonction du cours fait-elle référence ?
        \begin{solution}
            $k(t) = \Pi_\theta(t)$, le signal porte de largeur $\theta$ vu en cours (\S2).
        \end{solution}
        \item Rappeler la décomposition en série de Fourier de $x(t)$ en forme complexe. \textit{Indice: utiliser la formule d'Euler.}
        \begin{solution}
            Avec $\omega_0=\omega$, $\cos(\omega t) = \frac12e^{\complexj\omega t}+\frac12e^{-\complexj\omega t}$ (Euler) : c'est déjà la décomposition en série de Fourier complexe, avec $c_1=c_{-1}=\frac12$ et $c_n=0$ pour $n\notin\{-1,1\}$.
        \end{solution}
        \item $y(t)$ admet-il une décomposition en série de Fourier ? Justifier.
        \begin{solution}
            Non : $y$ est identiquement nulle en dehors de $[-\theta/2,\theta/2]$, donc n'est pas périodique (une fonction $T$-périodique non nulle ne peut pas s'annuler identiquement sur un intervalle infini). Les séries de Fourier (chapitre 1) ne s'appliquent qu'aux fonctions périodiques ; $y$, à support borné, relève de la transformée de Fourier (chapitre 2).
        \end{solution}
        \item Exprimer la transformée de Fourier de $x(t)$ à l'aide de deux impulsions de Dirac.

        \textit{Rappel :} 
        
        \textit{ $\mathcal F[\delta]=1$ (TF d'une impulsion) et par dualité, $\mathcal F[1]=\delta$.}
    
        \textit{$h(t)e^{2\complexj\pi f_0t} \overset{\mathcal F}{\longleftrightarrow} H(f-f_0)$.}

        \begin{solution}
            Avec $f_0=\omega/(2\pi)$, en remarquant que $x(t) = \frac{1}{2} e^{\complexj 2\pi f_0 t} + \frac{1}{2}e^{-\complexj 2\pi f_0 t}$, on trouve
            $$
                \mathcal F[x](f) = \frac12\delta(f-f_0) + \frac12\delta(f+f_0).
            $$
        \end{solution}
        \item Calculer l'expression de la transformée de Fourier de $k(t)$.
        \begin{solution}
            $k=\Pi_\theta$ est le signal porte, non nul seulement sur $[-\theta/2,\theta/2]$ :
            $$
                \mathcal F[k](f) = \int_{-\theta/2}^{\theta/2} e^{-2\complexj\pi ft}\,\d t
                = \left[\frac{e^{-2\complexj\pi ft}}{-2\complexj\pi f}\right]_{-\theta/2}^{\theta/2}
                = \frac{e^{-\complexj\pi f\theta} - e^{\complexj\pi f\theta}}{-2\complexj\pi f}
                = \frac{e^{\complexj\pi f\theta} - e^{-\complexj\pi f\theta}}{2\complexj\pi f}.
            $$
            Par la formule d'Euler, $e^{\complexj\pi f\theta} - e^{-\complexj\pi f\theta} = 2\complexj\sin(\pi f\theta)$, donc
            $$
                \mathcal F[k](f) = \frac{2\complexj\sin(\pi f\theta)}{2\complexj\pi f} = \frac{\sin(\pi f\theta)}{\pi f}
                = \theta\cdot\frac{\sin(\pi f\theta)}{\pi f\theta} = \theta\,\sinc(\pi f\theta).
            $$
        \end{solution}
        \item En déduire la transformée de Fourier de $y(t)$.

        \textit{Rappel : Un produit dans le domaine temporel se traduit par un produit de convolution dans le domaine fréquentiel}
        \begin{solution}
            
            $$
            Y(f) = \mathcal{F}(y)(f) = \mathcal{F}(k \times x)(f) = \mathcal{F}(k)(f) * \mathcal{F}(x)(f)
            $$

            D'après les questions précédentes, 
            $$
                \mathcal{F}(k)(f) = \theta\,\sinc(\pi f\theta),
            $$
            $$
                \mathcal{F}(x)(f) = \frac12\delta(f-f_0) + \frac12\delta(f+f_0).
            $$

            Ainsi, 

            $$
            \begin{aligned}
                  Y(f) &= \int_{-\infty}^{+\infty} K(u) X(u - f) \d u \\
                        &= \frac{1}{2} \int_{-\infty}^{+\infty} \theta\,\sinc(\pi f\theta) \delta(f-f_0) \d u + \frac{1}{2} \int_{-\infty}^{+\infty} \theta\,\sinc(\pi f\theta) \delta(f+f_0) \d u \\
                        &= \frac{\theta}{2} \left(\sinc(\pi (f-f_0)\theta) + \sinc(\pi (f+f_0)\theta)\right)
            \end{aligned}
            $$
        \end{solution}

    \end{enumerate}
\end{exercice}
