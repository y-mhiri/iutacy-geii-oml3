\begin{exercice}
    \label{ex:dualite-conv}

    \textit{Pour deux signaux $x,y:\R\to\R$ (des fonctions du temps $t$) tels que $x(t)\overset{\mathcal F}{\longleftrightarrow}X(f)$ et $y(t)\overset{\mathcal F}{\longleftrightarrow}Y(f)$ :
    $$
        X(f) = \int_{-\infty}^{+\infty} x(t)\,e^{-2\complexj\pi ft}\,\d t, \qquad\qquad (x*y)(t) = \int_{-\infty}^{+\infty} x(\tau)\,y(t-\tau)\,\d\tau.
    $$
    On admettra que, dans les calculs de cet exercice, l'ordre d'intégration d'une intégrale double peut être échangé, et qu'un changement de variable affine (par exemple $u=t-\tau$, à $\tau$ fixé) ne modifie pas des bornes $-\infty$ à $+\infty$.
    }

    \bigskip
    Le cours affirme, sans démonstration, que
    $$
    (x*y)(t) \overset{\mathcal F}{\longleftrightarrow} X(f)\,Y(f)
    $$
    (propriété de dualité produit/convolution). En utilisant les rappels ci-dessus, le démontrer.

    \medskip
    \noindent\textbf{Indication.} Pour tous $u,t\in\R$, $t = t+u-u$.
    \begin{solution}
        Par définition de la TF et du produit de convolution,
        $$
            \mathcal F[x*y](f) = \int_{-\infty}^{+\infty}\left(\int_{-\infty}^{+\infty} x(\tau)\,y(t-\tau)\,\d\tau\right) e^{-2\complexj\pi ft}\,\d t.
        $$
        En échangeant l'ordre d'intégration :
        $$
            \mathcal F[x*y](f) = \int_{-\infty}^{+\infty} x(\tau) \left(\int_{-\infty}^{+\infty} y(t-\tau)\,e^{-2\complexj\pi ft}\,\d t\right) \d\tau.
        $$
        Dans l'intégrale intérieure, on pose $u=t-\tau$ (donc $t=u+\tau$, $\d u=\d t$) : quand $t$ décrit $\R$ tout entier, $u$ aussi (pour tout $u\in\R$, $t=u+\tau$ convient), donc les bornes ne changent pas :
        $$
            \int_{-\infty}^{+\infty} y(t-\tau)\,e^{-2\complexj\pi ft}\,\d t = \int_{-\infty}^{+\infty} y(u)\,e^{-2\complexj\pi f(u+\tau)}\,\d u = e^{-2\complexj\pi f\tau}\int_{-\infty}^{+\infty} y(u)\,e^{-2\complexj\pi fu}\,\d u = e^{-2\complexj\pi f\tau}\,Y(f).
        $$
        D'où
        $$
            \mathcal F[x*y](f) = \int_{-\infty}^{+\infty} x(\tau)\,e^{-2\complexj\pi f\tau}\,Y(f)\,\d\tau = Y(f)\int_{-\infty}^{+\infty} x(\tau)\,e^{-2\complexj\pi f\tau}\,\d\tau = Y(f)\,X(f).
        $$
    \end{solution}
\end{exercice}

\begin{exercice}[Modulation d'un signal à bande limitée]
    Soit $x$ un signal (une fonction du temps $t\mapsto x(t)$) à bande limitée, c'est-à-dire dont la transformée de Fourier $X$ est nulle en dehors de $[-B,B]$ ($B>0$).
    
    Pour $f_0>B$, on pose
    $$
        m(t) = x(t)\sin(2\pi f_0t).
    $$
    \begin{enumerate}[\bfseries 1.]
        \item Exprimer $M=\mathcal F[m]$ en fonction de $X$.
        \begin{solution}
            Avec la formule d'Euler $\sin(2\pi f_0t) = \dfrac{e^{2\complexj\pi f_0t}-e^{-2\complexj\pi f_0t}}{2\complexj}$ et le décalage fréquentiel (vu en cours) :
            $$
                M(f) = \frac1{2\complexj}\Big(X(f-f_0) - X(f+f_0)\Big).
            $$
        \end{solution}

        \item En déduire le support de $M$, c'est-à-dire les valeurs de $f\in\R$ pour lesquelles $M(f)\neq0$.
        \begin{solution}
            $X$ est nulle en dehors de $[-B,B]$, donc $X(f-f_0)$ est nulle en dehors de $[f_0-B,f_0+B]$, et $X(f+f_0)$ en dehors de $[-f_0-B,-f_0+B]$. Comme $f_0>B$, ces deux intervalles sont disjoints, et le support de $M$ est leur réunion :
            $$
                [-f_0-B,\,-f_0+B]\ \cup\ [f_0-B,\,f_0+B].
            $$
            Moduler $x$ par un sinus translate donc son spectre autour de $\pm f_0$, sans en changer la forme.
        \end{solution}

        \item On remultiplie $m$ par le même sinus : $p(t) = m(t)\sin(2\pi f_0t)$. Exprimer $p(t)$ comme une somme de deux termes (transformation produit $\to$ somme).
        \begin{solution}
            $\sin^2(2\pi f_0t)$ se calcule avec la formule d'Euler :
            $$
                \sin^2(2\pi f_0t) = \left(\frac{e^{2\complexj\pi f_0t}-e^{-2\complexj\pi f_0t}}{2\complexj}\right)^{\!2} = \frac{e^{4\complexj\pi f_0t} - 2 + e^{-4\complexj\pi f_0t}}{-4} = \frac{1-\cos(4\pi f_0t)}2.
            $$
            D'où
            $$
                p(t) = x(t)\sin^2(2\pi f_0t) = \frac{x(t)}2 - \frac{x(t)\cos(4\pi f_0t)}2.
            $$
        \end{solution}

        \item Calculer $P=\mathcal F[p]$ en fonction de $X$.
        \begin{solution}
            Le premier terme a pour TF $\frac12X(f)$ par linéarité. Pour le second, avec la formule d'Euler $\cos(4\pi f_0t) = \frac12e^{4\complexj\pi f_0t}+\frac12e^{-4\complexj\pi f_0t}$ et le décalage fréquentiel (vu en cours, avec la fréquence $2f_0$ au lieu de $f_0$) :
            $$
                \mathcal F\big[x(t)\cos(4\pi f_0t)\big](f) = \frac12\Big(X(f-2f_0)+X(f+2f_0)\Big).
            $$
            Par linéarité,
            $$
                P(f) = \frac12X(f) - \frac14\Big(X(f-2f_0)+X(f+2f_0)\Big).
            $$
        \end{solution}

        \item Comment peut-on retrouver $x(t)$ à partir de $p(t)$ ? (Aucun calcul supplémentaire n'est demandé : justifier le principe à l'aide des questions précédentes.)
        \begin{solution}
            Comme pour $M$ (question 2), $f_0>B$ sépare les trois morceaux du spectre de $p$ : $\frac12X(f)$ occupe $[-B,B]$, tandis que les deux autres termes occupent $[2f_0-B,2f_0+B]$ et $[-2f_0-B,-2f_0+B]$, disjoints de $[-B,B]$ dès que $f_0>B$. Sur $[-B,B]$, on a donc exactement $P(f) = \frac12X(f)$.

            Un \highlight{filtre passe-bas idéal} qui laisse inchangées les fréquences de $[-B,B]$ et annule les autres, appliqué à $p$, donne donc $\frac12X(f)$ sur tout $\R$, soit $\frac12x(t)$ en temporel : on retrouve $x$ (à un facteur $2$ près) en multipliant par $2$ le résultat de ce filtrage. Moduler puis démoduler (remultiplier par la porteuse, puis filtrer passe-bas) permet ainsi de transmettre $x$ autour de $f_0$ et de le récupérer à l'arrivée.
        \end{solution}
    \end{enumerate}
\end{exercice}

\begin{exercice}[Convolution : deux portes]
    On note $\Pi_\theta$ la fonction porte de largeur $\theta>0$, centrée en $0$ :
    $$
        \Pi_\theta(t) = \begin{cases} 1 & \text{si } |t|<\theta/2 \\ 0 & \text{sinon.} \end{cases}
    $$
    \begin{enumerate}[\bfseries 1.]
        \item Tracer $\Pi_\theta$. Que vaut $\displaystyle\int_{-\infty}^{+\infty}\Pi_\theta(t)\,\d t$ ?
        \begin{solution}
            Un rectangle de largeur $\theta$ et de hauteur $1$. Son intégrale est l'aire de ce rectangle : $\theta$.
        \end{solution}

        \item Calculer $\mathcal F[\Pi_\theta]$.
        \begin{solution}
            $$
                \mathcal F[\Pi_\theta](f) = \int_{-\theta/2}^{\theta/2} e^{-2\complexj\pi ft}\,\d t
                = \left[\frac{e^{-2\complexj\pi ft}}{-2\complexj\pi f}\right]_{-\theta/2}^{\theta/2}
                = \frac{e^{\complexj\pi f\theta}-e^{-\complexj\pi f\theta}}{2\complexj\pi f}
                = \frac{\sin(\pi f\theta)}{\pi f} = \theta\,\sinc(\pi f\theta).
            $$
            En $f=0$, $\sinc(0)=1$ donc $\mathcal F[\Pi_\theta](0)=\theta$ : cohérent avec la question 1.
        \end{solution}

        \item Soit $x$ une fonction quelconque. Calculer $(x*\Pi_\theta)(t)$ et l'interpréter.
        \begin{solution}
            $\Pi_\theta(t-\tau)=1$ si et seulement si $|t-\tau|<\theta/2$, soit $\tau\in(t-\theta/2,\,t+\theta/2)$, et vaut $0$ sinon. Donc
            $$
                (x*\Pi_\theta)(t) = \int_{-\infty}^{+\infty} x(\tau)\,\Pi_\theta(t-\tau)\,\d\tau = \int_{t-\theta/2}^{t+\theta/2} x(\tau)\,\d\tau.
            $$
            C'est l'intégrale de $x$ sur une fenêtre de largeur $\theta$ centrée en $t$ : une aire glissante (une moyenne glissante, à un facteur $1/\theta$ près).
        \end{solution}

        \item Cas particulier $x=\Pi_\theta$ : calculer $(\Pi_\theta*\Pi_\theta)(t)$ et tracer le résultat.
        \begin{solution}
            D'après la question précédente,
            $$
                (\Pi_\theta*\Pi_\theta)(t) = \int_{t-\theta/2}^{t+\theta/2} \Pi_\theta(\tau)\,\d\tau,
            $$
            c'est-à-dire la longueur du recouvrement entre les intervalles $[-\theta/2,\theta/2]$ et $[t-\theta/2,t+\theta/2]$. Pour $|t|\geq\theta$, ces intervalles ne se recouvrent pas : le résultat est $0$. Pour $|t|<\theta$, le recouvrement vaut $\theta-|t|$. D'où
            $$
                (\Pi_\theta*\Pi_\theta)(t) = \begin{cases} \theta-|t| & \text{si } |t|<\theta \\ 0 & \text{sinon,} \end{cases}
            $$
            un triangle de base $2\theta$ et de hauteur $\theta$.
        \end{solution}

        \item En notant $Y=\mathcal F[\Pi_\theta*\Pi_\theta]$, le théorème produit-convolution donne $Y(f) = \mathcal F[\Pi_\theta](f)^2$. Sans calculer $Y$ entièrement, vérifier que $Y(0)$ est cohérent avec l'aire du triangle tracé à la question 4.
        \begin{solution}
            D'après la question 2, $\mathcal F[\Pi_\theta](0)=\theta$, donc $Y(0)=\theta^2$. Or $Y(0) = \displaystyle\int_{-\infty}^{+\infty}(\Pi_\theta*\Pi_\theta)(t)\,\d t$ est aussi l'aire du triangle de la question 4, qui vaut $\frac12\times(2\theta)\times\theta=\theta^2$. Les deux valeurs coïncident.
        \end{solution}
    \end{enumerate}
\end{exercice}

\begin{exercice}[Décomposition en portes décalées]
    On reprend la porte $\Pi_\theta$ définie dans l'exercice précédent, de transformée
    $\mathcal F[\Pi_\theta](f)=\theta\,\sinc(\pi f\theta)$.
    \begin{enumerate}[\bfseries 1.]
        \item Soit $x$ la fonction définie par $x(t)=A_1$ sur $[0,\theta[$, $A_2$ sur $[\theta,2\theta[$, $A_3$ sur $[2\theta,3\theta[$, et $0$ ailleurs, avec $A_1=1$, $A_2=2$, $A_3=-1$. Tracer son graphe et l'exprimer comme une somme de trois portes $\Pi_\theta$ centrées et décalées.
        \begin{solution}
            Chaque palier, de largeur $\theta$, est une porte $\Pi_\theta$ décalée pour être centrée en son milieu : le premier en $\theta/2$, le deuxième en $3\theta/2$, le troisième en $5\theta/2$. D'où
            $$
                f(t) = \Pi_\theta(t-\theta/2) + 2\,\Pi_\theta(t-3\theta/2) -\Pi_\theta(t-5\theta/2).
            $$
        \end{solution}

        \item En déduire $\mathcal F\{x\} (f)$ en reprenant les propriétés de la transformée de Fourier (linéarité, décalage).
        \begin{solution}
            Par linéarité, et en décalant $\Pi_\theta$ de $t_0$ (ce qui multiplie sa transformée par $e^{-2\complexj\pi ft_0}$) :
            $$
                \mathcal F[x](f) = \theta\,\sinc(\pi f\theta)\left(e^{-\complexj\pi f\theta} + 2\,e^{-3\complexj\pi f\theta} + e^{-5\complexj\pi f\theta}\right).
            $$
        \end{solution}

        \item Montrer que $|\mathcal F\{x\}(f)|=0$ en $f=k/\theta$ ($k\in\mathbb Z^*$), quels que soient $A_1,A_2,A_3$ et quelles que soient les positions des trois paliers.
        \begin{solution}
            D'après la question 2, $\mathcal F[x](f)$ s'écrit comme le produit de $\theta\,\sinc(\pi f\theta)$ par une somme de termes de la forme $A_k\,e^{-2\complexj\pi ft_k}$. Le premier facteur ne dépend ni des $A_k$ ni des positions $t_k$ : c'est le même dès que les trois portes ont la largeur $\theta$. Or un produit est nul dès que l'un de ses facteurs l'est : $\theta\,\sinc(\pi f\theta)=0$ pour $f=k/\theta$ ($k\in\mathbb Z^*$), donc $\mathcal F[x](f)=0$ en ces points, quelle que soit la valeur du second facteur.
        \end{solution}

        \item On reprend $(A_1,A_2,A_3)=(1,2,1)$, mais avec des largeurs différentes : $\theta_1=\theta$, $\theta_2=2\theta$, $\theta_3=3\theta$ (paliers toujours consécutifs). Exprimer $x$, puis $\mathcal F\{x\}$. L'argument de la question 3 s'applique-t-il encore ? Justifier.
        \begin{solution}
            Les paliers occupent maintenant $[0,\theta)$, $[\theta,3\theta)$, $[3\theta,6\theta)$, centrés respectivement en $\theta/2$, $2\theta$, $4{,}5\,\theta$ :
            $$
                x(t) = \Pi_\theta(t-\theta/2) + 2\,\Pi_{2\theta}(t-2\theta) + \Pi_{3\theta}(t-4{,}5\,\theta),
            $$
            $$
                \mathcal F[x](f) = \theta\,\sinc(\pi f\theta)\,e^{-\complexj\pi f\theta} + 4\theta\,\sinc(2\pi f\theta)\,e^{-4\complexj\pi f\theta} + 3\theta\,\sinc(3\pi f\theta)\,e^{-9\complexj\pi f\theta}.
            $$
            Non : chaque terme a maintenant sa propre largeur, donc son propre sinus cardinal. Il n'y a plus de facteur commun à mettre en évidence comme à la question 3 : l'argument de factorisation ne s'applique plus. Un zéro de $\mathcal F[x]$ résulterait ici d'une annulation particulière entre trois termes différents, pas d'une propriété structurelle garantie.
        \end{solution}
    \end{enumerate}
\end{exercice}
